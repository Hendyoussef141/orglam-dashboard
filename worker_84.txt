// --- BOSTA CUSTOMER RANKING INDEX ---
// Shopify and Bosta don't necessarily store phone numbers in the same format, and Bosta's own
// data has some malformed numbers too (seen in practice: "+2001277677762" - an extra 0 stuck
// between the 20 country code and the real number). This strips country-code/leading-zero
// noise repeatedly until it isolates the 10-digit Egyptian mobile core (e.g. "1277677762"),
// then returns that core plus its standard local/international forms so we can match reliably
// regardless of which messy format either system stored it in.
function phoneVariants(raw) {
  let d = (raw || "").replace(/[^\d]/g, "");
  if (!d) return [];
  while (d.length > 10 && (d.startsWith("20") || d.startsWith("0"))) {
    d = d.startsWith("20") ? d.slice(2) : d.slice(1);
  }
  if (d.length > 10) d = d.slice(-10); // fallback safety net
  const core = d;
  return [...new Set([core, "0" + core, "20" + core])];
}

// Bosta's /deliveries endpoint doesn't support filtering by receiverPhone, but it does return
// each delivery's receiver.ranking - so we page through recent deliveries once, index every
// receiver phone number (in all its formats) to their ranking, and cache that index in memory
// for a while so we're not re-paging Bosta on every single order lookup.
let bostaRankingIndexCache = null; // { builtAt, index }
let bostaRankingIndexBuildPromise = null;
const BOSTA_RANKING_INDEX_TTL_MS = 15 * 60 * 1000; // 15 minutes
async function getBostaRankingIndex(env) {
  if (bostaRankingIndexCache && (Date.now() - bostaRankingIndexCache.builtAt) < BOSTA_RANKING_INDEX_TTL_MS) {
    return bostaRankingIndexCache;
  }
  if (bostaRankingIndexBuildPromise) return bostaRankingIndexBuildPromise; // avoid duplicate concurrent builds
  bostaRankingIndexBuildPromise = (async () => {
    const BOSTA_BASE = "https://app.bosta.co";
    const authHeader = env.BOSTA_API_KEY;
    const index = {};
    const PAGE_SIZE = 100;
    const MAX_PAGES = 150;
    let pagesFetched = 0, deliveriesScanned = 0, lastError = null;
    // IMPORTANT: Bosta doesn't reliably honor our requested limit (confirmed: asking for
    // limit=1 still returned 10 results), so a page coming back smaller than PAGE_SIZE is
    // NOT a trustworthy "this is the last page" signal. We only stop early on a genuinely
    // empty page or a request failure - otherwise we always fetch up to MAX_PAGES.
    for (let page = 1; page <= MAX_PAGES; page++) {
      let res;
      try {
        res = await fetch(`${BOSTA_BASE}/api/v0/deliveries?limit=${PAGE_SIZE}&page=${page}`, {
          headers: { "Authorization": authHeader, "Content-Type": "application/json" },
        });
      } catch (e) { lastError = e.message; break; }
      if (!res.ok) { lastError = `HTTP ${res.status} on page ${page}`; break; }
      const json = await res.json();
      const rawList = json.data || json.deliveries || [];
      pagesFetched++;
      if (rawList.length === 0) break;
      deliveriesScanned += rawList.length;
      for (const d of rawList) {
        const phone = d.receiver?.phone;
        const ranking = d.receiver?.ranking;
        if (phone && ranking !== undefined && ranking !== null) {
          for (const v of phoneVariants(phone)) {
            if (!(v in index)) index[v] = ranking;
          }
        }
      }
    }
    bostaRankingIndexCache = { builtAt: Date.now(), index, pagesFetched, deliveriesScanned, lastError };
    return bostaRankingIndexCache;
  })();
  try {
    return await bostaRankingIndexBuildPromise;
  } finally {
    bostaRankingIndexBuildPromise = null;
  }
}

// Some SKUs reject labelOwner/prepOwner "SELLER" with e.g. "X does not require prepOwner but SELLER was
// assigned. Accepted values: [NONE]". Scans Amazon's error list and patches matching items in place;
// returns true if anything changed (caller should retry the request once).
function patchOwnerFieldsFromErrors(errors, itemsByMsku) {
  let changed = false;
  for (const err of errors || []) {
    const text = `${err.message || ""} ${err.details || ""}`;
    const re1 = /([\w-]+) does not require (\w+) but \w+ was assigned\. Accepted values: \[([A-Z_]+)\]/gi;
    let m;
    while ((m = re1.exec(text)) !== null) {
      const [, msku, field, accepted] = m;
      const list = itemsByMsku[msku];
      if (list) { list.forEach(it => { it[field] = accepted; }); changed = true; }
    }
    const re2 = /Invalid item details for ([\w-]+), expected: Item\([^)]*labelOwner=(\w+), prepOwner=(\w+)/gi;
    while ((m = re2.exec(text)) !== null) {
      const [, msku, labelOwner, prepOwner] = m;
      const list = itemsByMsku[msku];
      if (list) { list.forEach(it => { it.labelOwner = labelOwner; it.prepOwner = prepOwner; }); changed = true; }
    }
  }
  return changed;
}

// Exchange the long-lived refresh token for a short-lived access token
async function getAmazonAccessToken(env) {
  const res = await fetch("https://api.amazon.com/auth/o2/token", {
    method: "POST",
    headers: { "Content-Type": "application/x-www-form-urlencoded" },
    body: new URLSearchParams({
      grant_type: "refresh_token",
      refresh_token: env.LWA_REFRESH_TOKEN,
      client_id: env.LWA_CLIENT_ID,
      client_secret: env.LWA_CLIENT_SECRET,
    }),
  });
  const json = await res.json();
  if (!res.ok) throw new Error("Amazon token error: " + JSON.stringify(json));
  return json.access_token;
}

// Looks up a product's current Amazon listing price (your own offer, via the Product Pricing
// API), for estimating Pending Amazon orders that have no order-level price data yet. Uses a
// persistent KV cache so repeat SKUs cost zero extra calls once resolved.
async function getAmazonListingPrice(env, accessToken, sellerSku, priceCache) {
  if (!sellerSku) return null;
  const cached = priceCache[sellerSku];
  // Cached prices expire after 6 hours so they stay in sync as you change prices on Amazon,
  // instead of locking in whatever was first looked up.
  const PRICE_TTL_MS = 6 * 60 * 60 * 1000;
  if (cached !== undefined && cached !== null && typeof cached === "object" && (Date.now() - cached.fetchedAt) < PRICE_TTL_MS) {
    return cached.price;
  }
  try {
    const params = new URLSearchParams({ MarketplaceId: AMAZON_MARKETPLACE_ID, Skus: sellerSku, ItemType: "Sku" });
    const res = await fetch(`${AMAZON_ENDPOINT}/products/pricing/v0/price?${params.toString()}`, {
      headers: { "x-amz-access-token": accessToken },
    });
    const json = await res.json();
    if (!res.ok) return cached ? cached.price : null; // API hiccup - fall back to stale cache rather than nothing
    const offer = json.payload?.[0]?.Product?.Offers?.[0];
    const price = offer?.RegularPrice?.Amount ?? offer?.BuyingPrice?.ListingPrice?.Amount;
    const parsed = price != null ? parseFloat(price) : null;
    if (parsed !== null) priceCache[sellerSku] = { price: parsed, fetchedAt: Date.now() }; // don't lock in a failed lookup - retry next time
    return parsed !== null ? parsed : (cached ? cached.price : null);
  } catch (e) {
    return cached ? cached.price : null;
  }
}

// Batched version - resolves MANY SKUs in one Pricing API call instead of one call per SKU.
// Cuts subrequests way down (this app was hitting Cloudflare's per-request subrequest cap when
// pricing many Pending orders individually, which silently truncated the day's estimate).
async function getAmazonListingPricesBatch(env, accessToken, sellerSkus, priceCache) {
  const PRICE_TTL_MS = 6 * 60 * 60 * 1000;
  const result = {};
  const staleFallback = {}; // old cached price, used only if a live refetch fails
  const toFetch = [];
  for (const sku of sellerSkus) {
    const cached = priceCache[sku];
    if (cached && typeof cached === "object" && (Date.now() - cached.fetchedAt) < PRICE_TTL_MS) {
      result[sku] = cached.price;
    } else {
      if (cached && typeof cached === "object") staleFallback[sku] = cached.price;
      toFetch.push(sku);
    }
  }
  // Amazon's Get Pricing endpoint accepts up to 20 SKUs per call
  for (let i = 0; i < toFetch.length; i += 20) {
    const batch = toFetch.slice(i, i + 20);
    try {
      const params = new URLSearchParams({ MarketplaceId: AMAZON_MARKETPLACE_ID, ItemType: "Sku", Skus: batch.join(",") });
      const res = await fetch(`${AMAZON_ENDPOINT}/products/pricing/v0/price?${params.toString()}`, {
        headers: { "x-amz-access-token": accessToken },
      });
      const json = await res.json();
      if (res.ok) {
        for (const entry of (json.payload || [])) {
          const sku = entry.SellerSKU;
          const offer = entry.Product?.Offers?.[0];
          const price = offer?.RegularPrice?.Amount ?? offer?.BuyingPrice?.ListingPrice?.Amount;
          const parsed = price != null ? parseFloat(price) : null;
          if (sku && parsed !== null) {
            result[sku] = parsed;
            priceCache[sku] = { price: parsed, fetchedAt: Date.now() };
          }
        }
      } else {
        priceCache.__lastApiError = { status: res.status, body: JSON.stringify(json).slice(0, 300), at: Date.now() }; // TEMPORARY diagnostic
      }
    } catch (e) { priceCache.__lastApiError = { status: "fetch-threw", body: e.message, at: Date.now() }; /* leave unresolved SKUs in this batch as missing - stale/average fallback covers them below */ }
  }
  // Live refetch failed or returned nothing for these - use the old (stale) cached price
  // rather than dropping the SKU's price entirely, so today's orders still get priced.
  for (const sku of toFetch) {
    if (result[sku] === undefined && staleFallback[sku] !== undefined) {
      result[sku] = staleFallback[sku];
    }
  }
  return result;
}

// Real Amazon listing title + image per ASIN, via the Catalog Items API - the FBA
// inventory-summaries endpoint's own productName field is often a short/generic label, not the
// actual listing title, and Amazon items have nothing to do with the separate Shopify catalog,
// so this is the only trustworthy source for what a SKU is actually called/pictured on Amazon.
async function getAmazonCatalogInfo(env, accessToken, asins) {
  const CACHE_TTL_MS = 7 * 24 * 60 * 60 * 1000; // titles/images barely change - cache a week
  const cache = (await env.SALES_CACHE.get("amazon_catalog_info", { type: "json" })) || {};
  const result = {};
  const toFetch = [];
  for (const asin of asins) {
    const cached = cache[asin];
    if (cached && cached.image && (Date.now() - cached.fetchedAt) < CACHE_TTL_MS) result[asin] = cached;
    else toFetch.push(asin);
  }
  let changed = false;
  for (let i = 0; i < toFetch.length; i += 20) {
    const batch = toFetch.slice(i, i + 20);
    try {
      const params = new URLSearchParams({
        marketplaceIds: AMAZON_MARKETPLACE_ID,
        identifiers: batch.join(","),
        identifiersType: "ASIN",
        includedData: "summaries,images",
      });
      const res = await fetch(`${AMAZON_ENDPOINT}/catalog/2022-04-01/items?${params.toString()}`, {
        headers: { "x-amz-access-token": accessToken },
      });
      const json = await res.json();
      if (res.ok) {
        for (const entry of (json.items || [])) {
          const summary = entry.summaries?.[0];
          let image = null;
          for (const imgGroup of (entry.images || [])) {
            const imgs = imgGroup.images || (imgGroup.link ? [imgGroup] : []);
            image = imgs.find(im => im.variant === "MAIN") || imgs[0];
            if (image) break;
          }
          const info = { title: summary?.itemName || null, image: image?.link || null, fetchedAt: Date.now() };
          cache[entry.asin] = info;
          result[entry.asin] = info;
          changed = true;
        }
      }
      if (i + 20 < toFetch.length) await new Promise(r => setTimeout(r, 300));
    } catch (e) { /* leave these ASINs unresolved - existing name/image fallback covers them */ }
  }
  if (changed) await env.SALES_CACHE.put("amazon_catalog_info", JSON.stringify(cache));
  return result;
}

// Runs from the Cron Trigger every 5 min - only actually calls Amazon (and rewrites the cache)
// once the cache is 30+ min old, so opening the app never itself triggers an Amazon Orders API
// call and can't contribute to hitting its quota; the app always just reads whatever is cached.
async function refreshAmazonSalesCacheIfStale(env) {
  try {
    const cachedAmazon = await env.SALES_CACHE.get("amazon_sales_cache", { type: "json" });
    if (cachedAmazon && Date.now() - cachedAmazon.fetchedAt < 30 * 60 * 1000) return; // still fresh
    const amazonData = await getAmazonSales(env);
    await env.SALES_CACHE.put("amazon_sales_cache", JSON.stringify({ ...amazonData, fetchedAt: Date.now() }));
  } catch (e) { /* leave the existing cache in place - the next cron tick will retry */ }
}

// Fetch this month's orders from Amazon and sum today's + this month's totals
async function getAmazonSales(env) {
  const accessToken = await getAmazonAccessToken(env);

  const now = new Date();
  const cairoNow = new Date(now.getTime() + 3 * 60 * 60 * 1000);
  const startOfMonth = new Date(Date.UTC(cairoNow.getUTCFullYear(), cairoNow.getUTCMonth(), 1));
  const cairoDateStr = cairoNow.toISOString().split("T")[0];

  let allOrders = [];
  let nextToken = null;

  do {
    const params = new URLSearchParams({
      MarketplaceIds: AMAZON_MARKETPLACE_ID,
      CreatedAfter: startOfMonth.toISOString(),
    });
    if (nextToken) params.set("NextToken", nextToken);

    const res = await fetch(`${AMAZON_ENDPOINT}/orders/v0/orders?${params.toString()}`, {
      headers: { "x-amz-access-token": accessToken },
    });
    const json = await res.json();
    if (!res.ok) throw new Error("Amazon orders error: " + JSON.stringify(json));

    allOrders.push(...(json.payload?.Orders || []));
    nextToken = json.payload?.NextToken || null;
  } while (nextToken);

  let todayAmazonSales = 0, monthAmazonSales = 0, todayAmazonOrders = 0, monthAmazonOrders = 0, yesterdayAmazonSalesSoFar = 0;
  let skippedOrders = []; // orders still excluded this run (backlog not yet caught up)
  const byDate = {}; // { 'YYYY-MM-DD': { sales, orders } } - built from the same pass, no extra API calls

  // Amazon's Seller Central "Sales" figure includes ALL placed orders (even unconfirmed ones),
  // but the Orders API's OrderTotal field only populates once an order is fully confirmed - this
  // gap can span the WHOLE month, not just today, and can be dozens of orders wide. Resolving all
  // of them via item-level pricing in one run would blow through this app's very small quota, so
  // instead: permanently cache each order's resolved amount once looked up (never re-fetch the
  // same order twice), and only resolve a small, safe number of NEW ones per run - the backlog
  // closes gradually over several dashboard loads instead of one risky burst.
  const resolvedCache = (await env.SALES_CACHE.get("amazon_resolved_amounts", { type: "json" })) || {};
  // Manual price list the user maintains in-app (Amazon Prices screen) while the Pricing API
  // role is still pending Amazon's approval - checked first, before any live API lookup.
  // Last-resort fallback average, so a pending order is never fully dropped from the totals
  // just because its SKU has no live Pricing API result yet - Seller Central's "Sales" figure
  // counts every placed order, so we should never silently exclude one for pricing reasons alone.
  // Manual pricing was removed; the average is now derived from whatever live prices we've
  // already resolved and cached for other SKUs this run/recently.
  const skuPriceCacheForAvg = (await env.SALES_CACHE.get("amazon_sku_price_estimates", { type: "json" })) || {};
  const manualPrices = (await env.SALES_CACHE.get("amazon_manual_prices", { type: "json" })) || {};
  const knownPrices = Object.values(skuPriceCacheForAvg).map(v => v?.price).concat(Object.values(manualPrices)).filter(p => typeof p === "number" && p > 0);
  const globalAvgAmazonPrice = knownPrices.length ? knownPrices.reduce((a, b) => a + b, 0) / knownPrices.length : null;
  // One-time cleanup: earlier code could wrongly cache $0 for Pending orders with no price
  // allocated yet. Purge those zero entries so they get re-resolved for real below.
  let purgedZeros = false;
  for (const key of Object.keys(resolvedCache)) {
    if (resolvedCache[key] === 0) { delete resolvedCache[key]; purgedZeros = true; }
  }
  let newLookups = 0;
  // No cap for TODAY's or YESTERDAY's orders - both must resolve fully so their totals match
  // Seller Central exactly (yesterday's figure is still actively compared against once the day
  // rolls over). The cap only throttles genuinely old backlog (2+ days), which can wait a few runs.
  const MAX_NEW_BACKLOG_LOOKUPS_PER_RUN = 10;
  let newBacklogLookups = 0;
  const yesterdayDateStrForCap = new Date(cairoNow.getTime() - 24 * 60 * 60 * 1000).toISOString().split("T")[0];
  let resolvedCacheChanged = purgedZeros;
  const skuPriceCache = (await env.SALES_CACHE.get("amazon_sku_price_estimates", { type: "json" })) || {};
  let skuPriceCacheChanged = false;
  let estimatedOrderIds = [];

  // Process today's orders FIRST, so the (more time-sensitive) "today" figure catches up
  // quickly, rather than spending the whole per-run lookup budget on older backlog orders
  // first (which would leave "today" wrong for potentially many hours).
  const sortedOrders = [...allOrders].sort((a, b) => {
    const aPurchase = new Date(a.PurchaseDate);
    const aCairo = new Date(aPurchase.getTime() + 3 * 60 * 60 * 1000);
    const aDateStr = aCairo.toISOString().split("T")[0];
    const aIsToday = aDateStr === cairoDateStr || aDateStr === yesterdayDateStrForCap;
    const bPurchase = new Date(b.PurchaseDate);
    const bCairo = new Date(bPurchase.getTime() + 3 * 60 * 60 * 1000);
    const bDateStr = bCairo.toISOString().split("T")[0];
    const bIsToday = bDateStr === cairoDateStr || bDateStr === yesterdayDateStrForCap;
    if (aIsToday === bIsToday) return 0;
    return aIsToday ? -1 : 1; // today's/yesterday's orders sort first
  });

  for (const order of sortedOrders) {
    if (order.OrderStatus === "Canceled") continue;

    let amount = order.OrderTotal ? parseFloat(order.OrderTotal.Amount || 0) : null;

    if (amount === null && resolvedCache[order.AmazonOrderId] !== undefined) {
      amount = resolvedCache[order.AmazonOrderId];
    }

    const purchaseDateForCap = new Date(order.PurchaseDate);
    const cairoOrderTimeForCap = new Date(purchaseDateForCap.getTime() + 3 * 60 * 60 * 1000);
    const orderDateStrForCap = cairoOrderTimeForCap.toISOString().split("T")[0];
    const isTodayOrder = orderDateStrForCap === cairoDateStr || orderDateStrForCap === yesterdayDateStrForCap;

    if (amount === null && (isTodayOrder || newBacklogLookups < MAX_NEW_BACKLOG_LOOKUPS_PER_RUN)) {
      newLookups++;
      if (!isTodayOrder) newBacklogLookups++;
      if (newLookups > 1) await new Promise(resolve => setTimeout(resolve, 300)); // small gap between lookups to stay under Amazon's rate limit
      try {
        const itemsRes = await fetch(`${AMAZON_ENDPOINT}/orders/v0/orders/${order.AmazonOrderId}/orderItems`, {
          headers: { "x-amz-access-token": accessToken },
        });
        const itemsJson = await itemsRes.json();
        if (itemsRes.ok) {
          const items = itemsJson.payload?.OrderItems || [];
          const anyPricing = items.some(item => item.ItemPrice !== undefined);
          if (anyPricing) {
            amount = items.reduce((sum, item) => {
              const itemPrice = parseFloat(item.ItemPrice?.Amount || 0);
              const itemTax = parseFloat(item.ItemTax?.Amount || 0);
              return sum + itemPrice + itemTax;
            }, 0);
            resolvedCache[order.AmazonOrderId] = amount;
            resolvedCacheChanged = true;
          } else {
            // Pending order - Amazon hasn't allocated real pricing yet. Estimate using each
            // item's current LIVE Amazon listing price so estimates track real price changes
            // over time; falls back to your manual price list only if the Pricing API has
            // nothing for this SKU. This estimate is NOT written to resolvedCache, so the real
            // confirmed amount still takes over automatically once Amazon provides it.
            let estimatedTotal = 0;
            let allSkusPriced = true;
            const skusInOrder = [...new Set(items.map(item => item.SellerSKU).filter(Boolean))];
            const batchPrices = await getAmazonListingPricesBatch(env, accessToken, skusInOrder, skuPriceCache);
            skuPriceCacheChanged = true;
            for (const item of items) {
              const sku = item.SellerSKU;
              const asin = item.ASIN;
              const qty = parseInt(item.QuantityOrdered || 1, 10);
              if (!sku && !asin) { allSkusPriced = false; continue; }
              let price = sku && manualPrices[sku] > 0 ? manualPrices[sku] : (sku ? batchPrices[sku] : undefined);
              skuPriceCacheChanged = true;
              if ((price === null || price === undefined) && globalAvgAmazonPrice !== null) {
                price = globalAvgAmazonPrice; // no manual/live price for this item - use fallback rather than dropping the whole order
              }
              if (price === null || price === undefined) { allSkusPriced = false; continue; }
              estimatedTotal += price * qty;
            }
            if (estimatedTotal > 0) {
              amount = estimatedTotal; // use what we could price, even if one line item's SKU/ASIN had no match anywhere - partial is closer to truth than dropping the order to 0
              estimatedOrderIds.push(order.AmazonOrderId);
            }
          }
        }
      } catch (e) { /* leave amount as null if this lookup fails - fallback below covers it */ }
    }

    // Today's/yesterday's orders must never be silently dropped - Seller Central's "Sales"
    // figure counts every placed order regardless of whether we've resolved a real price yet.
    // Fall back to the global average price rather than excluding the order from the total.
    if (amount === null && globalAvgAmazonPrice !== null) {
      amount = globalAvgAmazonPrice;
      estimatedOrderIds.push(order.AmazonOrderId);
    }

    if (amount === null) {
      // No price resolved and no average yet to fall back on (e.g. very first run) - still count
      // the order as zero-value rather than dropping it, so order counts stay accurate.
      skippedOrders.push({ id: order.AmazonOrderId, status: order.OrderStatus, purchaseDate: order.PurchaseDate });
      amount = 0;
    }

    const purchaseDate = new Date(order.PurchaseDate);
    const cairoOrderTime = new Date(purchaseDate.getTime() + 3 * 60 * 60 * 1000);
    const orderDateStr = cairoOrderTime.toISOString().split("T")[0];

    monthAmazonSales += amount;
    monthAmazonOrders += 1;
    if (!byDate[orderDateStr]) byDate[orderDateStr] = { sales: 0, orders: 0 };
    byDate[orderDateStr].sales += amount;
    byDate[orderDateStr].orders += 1;
    if (orderDateStr === cairoDateStr) {
      todayAmazonSales += amount;
      todayAmazonOrders += 1;
    }
    // Same-time-of-day comparison for yesterday, mirroring the Shopify trend badge logic -
    // only counts yesterday's Amazon sales up to the same hour:minute as right now.
    const yesterdayDateStrForCutoff = new Date(cairoNow.getTime() - 24 * 60 * 60 * 1000).toISOString().split("T")[0];
    if (orderDateStr === yesterdayDateStrForCutoff && (cairoOrderTime.getUTCHours() < cairoNow.getUTCHours() || (cairoOrderTime.getUTCHours() === cairoNow.getUTCHours() && cairoOrderTime.getUTCMinutes() <= cairoNow.getUTCMinutes()))) {
      yesterdayAmazonSalesSoFar += amount;
    }
  }

  if (resolvedCacheChanged) {
    await env.SALES_CACHE.put("amazon_resolved_amounts", JSON.stringify(resolvedCache));
  }
  if (skuPriceCacheChanged) {
    await env.SALES_CACHE.put("amazon_sku_price_estimates", JSON.stringify(skuPriceCache));
  }

  return { todayAmazonSales, monthAmazonSales, todayAmazonOrders, monthAmazonOrders, skippedOrders, totalOrdersFetched: allOrders.length, resolvedBacklogSize: Object.keys(resolvedCache).length, knownPricesCount: knownPrices.length, pricingApiLastError: skuPriceCache.__lastApiError || null, byDate, estimatedOrderIds, yesterdayAmazonSales: byDate[new Date(cairoNow.getTime() - 24 * 60 * 60 * 1000).toISOString().split("T")[0]]?.sales || 0, yesterdayAmazonSalesSoFar };
}

// Convert a PEM-formatted private key into a CryptoKey for signing
async function importNoonPrivateKey(pem) {
  const pemContents = pem
    .trim()
    .replace(/^"+|"+$/g, "") // strip leading/trailing quote marks if accidentally included
    .replace(/,\s*$/g, "") // strip a trailing comma if accidentally included
    .replace(/\\n/g, "\n") // handle literal backslash-n sequences from JSON-style paste
    .replace("-----BEGIN PRIVATE KEY-----", "")
    .replace("-----END PRIVATE KEY-----", "")
    .replace(/\s/g, "");
  const binaryDer = Uint8Array.from(atob(pemContents), (c) => c.charCodeAt(0));

  return await crypto.subtle.importKey(
    "pkcs8",
    binaryDer,
    { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
    false,
    ["sign"]
  );
}

// Base64url encode helper (JWTs use base64url, not plain base64)
function base64url(input) {
  let base64;
  if (typeof input === "string") {
    base64 = btoa(input);
  } else {
    base64 = btoa(String.fromCharCode(...new Uint8Array(input)));
  }
  return base64.replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

// Build and sign the RS256 JWT noon expects: { sub, iat, jti }
async function createNoonJwt(env) {
  const header = { alg: "RS256", typ: "JWT" };
  const payload = {
    sub: env.NOON_KEY_ID,
    iat: Math.floor(Date.now() / 1000),
    jti: crypto.randomUUID(),
  };

  const encodedHeader = base64url(JSON.stringify(header));
  const encodedPayload = base64url(JSON.stringify(payload));
  const signingInput = `${encodedHeader}.${encodedPayload}`;

  const privateKey = await importNoonPrivateKey(env.NOON_PRIVATE_KEY);
  const signature = await crypto.subtle.sign(
    "RSASSA-PKCS1-v1_5",
    privateKey,
    new TextEncoder().encode(signingInput)
  );

  return `${signingInput}.${base64url(signature)}`;
}

// Log in to noon and return the session cookie string to reuse on later requests

    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "User-Agent": NOON_USER_AGENT,
    },
    body: JSON.stringify({
      token: jwt,
      default_project_code: env.NOON_PROJECT_CODE,
    }),
  });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`Noon login failed: ${res.status} ${text}`);
  }

    ? res.headers.getSetCookie()
    : [res.headers.get("set-cookie")].filter(Boolean);

  if (!setCookieHeaders.length) {
    throw new Error("Noon login succeeded but no session cookie was returned");
  }

  const cookieString = setCookieHeaders.map((c) => c.split(";")[0]).join("; ");
  return cookieString;
}

// Recursively search a parsed JSON value for the first object whose sku/partner_sku/code
// field case-insensitively matches targetSku, to pull out psku_code/cubic_feet generically
// (the exact response shape of list_eligible_lines hasn't been confirmed yet).
function findSkuLine(node, targetSku, seen = new Set()) {
  if (!node || typeof node !== "object" || seen.has(node)) return null;
  seen.add(node);
  const target = compactAlnumWorker(targetSku);
  const skuFields = ["sku", "partner_sku", "psku", "code", "sku_canonical"];
  if (skuFields.some(f => typeof node[f] === "string" && compactAlnumWorker(node[f]) === target)) {
    return node;
  }
  for (const key of Object.keys(node)) {
    const found = findSkuLine(node[key], targetSku, seen);
    if (found) return found;
  }
  return null;
}

// Create an export job and poll until it's ready, returning the download URL
async function noonRunExport(env, categoryCode, params, maxPolls = 55, existingCookie = null) {
  const cookie = existingCookie || (await getNoonSessionCookie(env));

  const createRes = await fetch(`${NOON_BASE_URL}/impex/v1/export/create`, {
    method: "POST",
    headers: { "Content-Type": "application/json", "User-Agent": NOON_USER_AGENT, "Cookie": cookie },
    body: JSON.stringify({ export_category_code: categoryCode, params }),
  });
  const createJson = await createRes.json();
  if (!createRes.ok) throw new Error("Noon export create error: " + JSON.stringify(createJson));

  const exportCode = createJson.export_code;
  let statusJson = null;

  for (let i = 0; i < maxPolls; i++) {
    const statusRes = await fetch(`${NOON_BASE_URL}/impex/v1/export/status`, {
      method: "POST",
      headers: { "Content-Type": "application/json", "User-Agent": NOON_USER_AGENT, "Cookie": cookie },
      body: JSON.stringify({ export_code: exportCode }),
    });
    statusJson = await statusRes.json();
    if (statusJson.export_status === "COMPLETE" || statusJson.export_status === "ERROR") break;
    await new Promise((resolve) => setTimeout(resolve, 2000));
  }

  if (!statusJson || statusJson.export_status !== "COMPLETE") {
    throw new Error("Noon is still preparing this export - it's taking longer than usual. Please try again in a minute.");
  }

  const fileRes = await fetch(statusJson.download_url);
  return await fileRes.text();
}

// Minimal CSV line parser that handles quoted fields containing commas
function parseCsvLine(line) {
  const fields = [];
  let current = "";
  let inQuotes = false;

  for (let i = 0; i < line.length; i++) {
    const char = line[i];
    if (char === '"') {
      inQuotes = !inQuotes;
    } else if (char === "," && !inQuotes) {
      fields.push(current);
      current = "";
    } else {
      current += char;
    }
  }
  fields.push(current);
  return fields;
}

// Fetch this month's finance transaction report and sum sales by Order Date
async function getNoonSales(env) {
  const now = new Date();
  const cairoNow = new Date(now.getTime() + 3 * 60 * 60 * 1000);
  const startOfMonth = new Date(Date.UTC(cairoNow.getUTCFullYear(), cairoNow.getUTCMonth(), 1));
  const cairoDateStr = cairoNow.toISOString().split("T")[0];

  // Rolling 3-day window (today, yesterday, day before) - since this report can lag
  // a couple of days, summing a small window is much less likely to show a misleading
  // zero than checking a single fixed day.
  const recentWindowStart = new Date(cairoNow.getTime() - 2 * 24 * 60 * 60 * 1000);
  const recentDateStrs = new Set([0, 1, 2].map((d) => new Date(recentWindowStart.getTime() + d * 24 * 60 * 60 * 1000).toISOString().split("T")[0]));

  // Small buffer before month start so the recent-window check still works correctly
  // right at the start of a new month.
  const lookbackStart = new Date(startOfMonth.getTime() - 5 * 24 * 60 * 60 * 1000);
  const fromDate = lookbackStart.toISOString().split("T")[0];
  const toDate = cairoNow.toISOString().split("T")[0];

  const cookie = await getNoonSessionCookie(env);

  const csvText = await noonRunExport(env, "noon_catalog_reports_productviewsandsalesdata", {
    country: "eg",
    lang: "en",
    from_date: fromDate,
    to_date: toDate,
  }, 16, cookie);

  const lines = csvText.trim().split("\n");
  const header = parseCsvLine(lines[0]);
  const visitDateIdx = header.indexOf("Visit_Date");
  const revenueShippedIdx = header.indexOf("Revenue_Shipped");
  const skuIdx = header.indexOf("SKU");
  const partnerSkuIdx = header.indexOf("Partner_SKU");
  const shippedUnitsIdx = header.indexOf("Shipped_Units");

  let monthNoonSales = 0, monthNoonOrders = 0;
  let recentNoonSales = 0, recentNoonOrders = 0;
  const currentMonthPrefix = cairoNow.toISOString().slice(0, 7);

  // Accumulate revenue/units per SKU so we can derive an average recent
  // selling price - used below as a fallback price source.
  const priceAccum = {};
  // Map from Noon's internal SKU to your own Partner SKU - needed since the live
  // catalog price lookup (GetProductOffers) takes partner_sku, not Noon's sku.
  const skuToPartnerSku = {};

  // Track real revenue per exact calendar day (not an estimate - this is Noon's own
  // reported figure). Used below to prefer exact data over the order-count estimate
  // whenever this report has actually caught up on a given day.
  const realRevenueByDate = {};

  for (let i = 1; i < lines.length; i++) {
    if (!lines[i]) continue;
    const fields = parseCsvLine(lines[i]);

    const visitDateStr = (fields[visitDateIdx] || "").trim();
    if (!visitDateStr) continue;

    const revenue = parseFloat(fields[revenueShippedIdx] || "");
    const units = parseFloat(fields[shippedUnitsIdx] || "");
    const sku = (fields[skuIdx] || "").trim();
    const partnerSku = (fields[partnerSkuIdx] || "").trim();

    if (sku && partnerSku && !skuToPartnerSku[sku]) skuToPartnerSku[sku] = partnerSku;

    if (sku && revenue > 0 && units > 0) {
      if (!priceAccum[sku]) priceAccum[sku] = { revenue: 0, units: 0 };
      priceAccum[sku].revenue += revenue;
      priceAccum[sku].units += units;
    }

    if (!revenue) continue; // skip rows with no shipped revenue for the totals below

    realRevenueByDate[visitDateStr] = (realRevenueByDate[visitDateStr] || 0) + revenue;

    // Recent-window check is independent of the current-month filter, so it still
    // works correctly right at the start of a new month
    if (recentDateStrs.has(visitDateStr)) {
      recentNoonSales += revenue;
      recentNoonOrders += 1;
    }

    // Only count towards month totals for activity within the current month
    if (!visitDateStr.startsWith(currentMonthPrefix)) continue;

    monthNoonSales += revenue;
    monthNoonOrders += 1;
  }

  const avgPriceBySku = {};
  for (const sku in priceAccum) {
    avgPriceBySku[sku] = priceAccum[sku].revenue / priceAccum[sku].units;
  }

  // Global fallback price (overall average across all known SKUs) - used when a specific
  // SKU has no recent price history AND no persistent catalog price cached yet.
  let totalRevenueAllSkus = 0, totalUnitsAllSkus = 0;
  for (const sku in priceAccum) {
    totalRevenueAllSkus += priceAccum[sku].revenue;
    totalUnitsAllSkus += priceAccum[sku].units;
  }
  const globalAvgPrice = totalUnitsAllSkus > 0 ? totalRevenueAllSkus / totalUnitsAllSkus : 0;

  // Persistent catalog price cache - grows over time as different SKUs get looked up,
  // so most refreshes eventually need zero new live lookups once the cache is "warm".
  let priceCache = {};
  try {
    priceCache = (await env.SALES_CACHE.get("noon_price_cache", { type: "json" })) || {};
  } catch (e) {
    priceCache = {};
  }

  // --- Estimate today's sales, and find the best available "yesterday" figure ---
  let todayNoonSales = 0, todayNoonOrders = 0;
  let recentRealSales = 0, recentRealOrders = 0, recentRealDate = null, recentRealIsEstimate = false, recentRealSalesSoFar = 0;
  let unpricedItemCount = 0;
  let todayEstimateFailed = false;
  try {
    const yesterdayDateStr = new Date(cairoNow.getTime() - 24 * 60 * 60 * 1000).toISOString().split("T")[0];
    const estimates = await getNoonRecentDaysEstimate(
      env,
      { avgPriceBySku, globalAvgPrice, skuToPartnerSku, priceCache, lookupBudget: 10 },
      yesterdayDateStr,
      cairoDateStr,
      cookie
    );
    todayNoonSales = estimates[cairoDateStr]?.sales || 0;
    todayNoonOrders = estimates[cairoDateStr]?.orders || 0;

    if (realRevenueByDate[yesterdayDateStr] !== undefined) {
      // Best case: Noon's own report has already caught up on yesterday specifically
      recentRealDate = yesterdayDateStr;
      recentRealSales = realRevenueByDate[yesterdayDateStr];
      recentRealOrders = estimates[yesterdayDateStr]?.orders || null;
      recentRealIsEstimate = false;
      // That report has no per-order timestamps, but the order-level estimate report does -
      // use it just for the same-time-of-day comparison value.
      recentRealSalesSoFar = estimates[yesterdayDateStr]?.salesSoFar || 0;
    } else if (estimates[yesterdayDateStr]?.sales > 0 || estimates[yesterdayDateStr]?.orders > 0) {
      // Report hasn't caught up yet - use our own live estimate for yesterday instead of
      // showing an older real figure. This is what "today" naturally becomes once the
      // clock passes midnight, since it's the same order-count-based estimate.
      recentRealDate = yesterdayDateStr;
      recentRealSales = estimates[yesterdayDateStr].sales;
      recentRealOrders = estimates[yesterdayDateStr].orders;
      recentRealIsEstimate = true;
      recentRealSalesSoFar = estimates[yesterdayDateStr].salesSoFar || 0;
    } else {
      // Last resort: search further back through real data
      for (let daysBack = 2; daysBack <= 14; daysBack++) {
        const candidateDateStr = new Date(cairoNow.getTime() - daysBack * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
        if (realRevenueByDate[candidateDateStr] !== undefined) {
          recentRealDate = candidateDateStr;
          recentRealSales = realRevenueByDate[candidateDateStr];
          recentRealOrders = estimates[candidateDateStr]?.orders || null;
          recentRealIsEstimate = false;
          recentRealSalesSoFar = estimates[candidateDateStr]?.salesSoFar || 0;
          break;
        }
      }
    }

    unpricedItemCount = estimates._unpricedItemCount || 0;

    // Save any newly-learned catalog prices back to KV for future runs to reuse
    if (estimates._lookupsUsed > 0) {
      await env.SALES_CACHE.put("noon_price_cache", JSON.stringify(priceCache));
    }
  } catch (e) {
    console.log("Noon recent-days estimate failed:", e.message);
    // This secondary export failed (e.g. quota exceeded) - flag it so the caller can
    // preserve the last-known-good today/recent figures instead of overwriting them with 0.
    todayEstimateFailed = true;
  }

  return { todayNoonSales, monthNoonSales, todayNoonOrders, monthNoonOrders, recentNoonSales, recentNoonOrders, recentRealSales, recentRealOrders, recentRealDate, recentRealIsEstimate, recentRealSalesSoFar, unpricedItemCount, todayEstimateFailed };
}

// Look up the current live price for a single SKU via Noon's catalog Offer API,
// using your Partner SKU (the endpoint doesn't accept Noon's internal SKU format).
async function getNoonOfferPrice(cookie, partnerSku) {
  const res = await fetch(`${NOON_BASE_URL}/offer/v1/product/${encodeURIComponent(partnerSku)}`, {
    headers: { "User-Agent": NOON_USER_AGENT, "Cookie": cookie },
  });
  if (!res.ok) return null;
  const json = await res.json();
  const offers = json.offers || [];
  const egyptOffer = offers.find((o) => o.country_code === "eg") || offers[0];
  if (!egyptOffer || !egyptOffer.price) return null;
  const p = egyptOffer.price;
  const amount = p.amount ?? p.sale_price ?? p.value ?? p.selling_price ?? (typeof p === "number" ? p : null);
  const parsed = amount != null ? parseFloat(amount) : null;
  return parsed && parsed > 0 ? parsed : null;
}

// Fetch orders across a small date range (near-real-time, no settlement lag) and estimate
// their EGP value per calendar day. Price for each item is resolved in this priority order:
// 1) persistent catalog price cache (from a real live lookup, most accurate)
// 2) recent average selling price for that SKU (from the sales report)
// 3) global average price across all known SKUs (last-resort fallback)
// A limited number of unknown SKUs get a real live lookup each run, which then gets
// saved into the persistent cache so future runs need fewer (ideally zero) new lookups.
async function getNoonRecentDaysEstimate(env, priceContext, fromDateStr, toDateStr, cookie) {
  const { avgPriceBySku, globalAvgPrice, skuToPartnerSku, priceCache, lookupBudget } = priceContext;

  const csvText = await noonRunExport(env, "noon_noonoms_ordersexport", {
    from_date: fromDateStr,
    to_date: toDateStr,
  }, 14, cookie);

  const lines = csvText.trim().split("\n");
  const header = parseCsvLine(lines[0]);
  const skuIdx = header.indexOf("sku");
  const statusIdx = header.indexOf("item_status");
  const orderNrIdx = header.indexOf("order_nr");
  const placedAtIdx = header.indexOf("order_placed_at");

  // First pass: collect valid rows (including cancelled, tracked via a flag) and find
  // which SKUs have no known price anywhere yet
  const rows = [];
  const skusNeedingLookup = new Set();

  for (let i = 1; i < lines.length; i++) {
    if (!lines[i]) continue;
    const fields = parseCsvLine(lines[i]);

    const status = (fields[statusIdx] || "").toLowerCase();
    const isCancelled = status === "cancelled";

    const placedAtRaw = (fields[placedAtIdx] || "").trim();
    if (!placedAtRaw) continue;
    // Format observed: "2026-07-01 01:43:16 UTC" - convert to a real Date, then to Cairo local date
    const placedAtDate = new Date(placedAtRaw.replace(" UTC", "Z").replace(" ", "T"));
    if (isNaN(placedAtDate.getTime())) continue;
    const cairoPlacedAt = new Date(placedAtDate.getTime() + 3 * 60 * 60 * 1000);
    const dateStr = cairoPlacedAt.toISOString().split("T")[0];

    const sku = (fields[skuIdx] || "").trim();
    const orderNr = fields[orderNrIdx];
    rows.push({ dateStr, sku, orderNr, isCancelled, hour: cairoPlacedAt.getUTCHours(), minute: cairoPlacedAt.getUTCMinutes() });

    if (sku && priceCache[sku] === undefined) skusNeedingLookup.add(sku);
  }

  // Live lookups for a limited number of unknown SKUs (respecting the subrequest budget)
  let lookupsUsed = 0;
  for (const sku of skusNeedingLookup) {
    if (lookupsUsed >= lookupBudget) break;
    const partnerSku = skuToPartnerSku[sku];
    if (!partnerSku) continue; // no mapping yet - will be resolved once seen in the sales report
    lookupsUsed += 1;
    try {
      const price = await getNoonOfferPrice(cookie, partnerSku);
      if (price) priceCache[sku] = price;
    } catch (e) {
      // Ignore individual lookup failures - falls through to the average-price fallback below
    }
  }

  // Second pass: compute sums using the best available price per SKU - both excluding
  // cancelled items (our normal definition) AND including everything (in case Noon's own
  // dashboard counts gross bookings regardless of later cancellation - this is temporarily
  // tracked side by side so we can compare against Noon's own displayed figure directly).
  const byDate = {};
  let unpricedItemCount = 0;
  const nowCairo = new Date(Date.now() + 3 * 60 * 60 * 1000);
  const nowHour = nowCairo.getUTCHours(), nowMinute = nowCairo.getUTCMinutes();

  for (const row of rows) {
    if (!byDate[row.dateStr]) byDate[row.dateStr] = { sales: 0, salesInclCancelled: 0, orderNrs: new Set(), orderNrsInclCancelled: new Set(), salesSoFar: 0 };

    let price = priceCache[row.sku];
    if (price === undefined) price = avgPriceBySku[row.sku];
    if (price === undefined) {
      price = globalAvgPrice;
      unpricedItemCount += 1;
    }

    byDate[row.dateStr].salesInclCancelled += price;
    if (row.orderNr) byDate[row.dateStr].orderNrsInclCancelled.add(row.orderNr);

    if (!row.isCancelled) {
      byDate[row.dateStr].sales += price;
      if (row.orderNr) byDate[row.dateStr].orderNrs.add(row.orderNr);
      // Same-time-of-day comparison: only count this order if it happened by the same
      // hour:minute as right now, so a full previous day isn't compared against today's
      // still-partial day (which would always look like a decline).
      if (row.hour < nowHour || (row.hour === nowHour && row.minute <= nowMinute)) {
        byDate[row.dateStr].salesSoFar += price;
      }
    }
  }

  const result = {};
  for (const dateStr in byDate) {
    result[dateStr] = {
      sales: byDate[dateStr].sales,
      orders: byDate[dateStr].orderNrs.size,
      salesInclCancelled: byDate[dateStr].salesInclCancelled,
      ordersInclCancelled: byDate[dateStr].orderNrsInclCancelled.size,
      salesSoFar: byDate[dateStr].salesSoFar,
    };
  }
  result._unpricedItemCount = unpricedItemCount;
  result._lookupsUsed = lookupsUsed;
  return result;
}

// Run the (slow) Noon export and save the result into KV for /sales to read instantly
async function refreshNoonCache(env) {
  try {
    const noonData = await getNoonSales(env);

    if (noonData.todayEstimateFailed) {
      // The secondary export (today/recent-day estimate) failed - preserve whatever
      // today/recent figures were already cached instead of overwriting them with 0,
      // while still letting the month total (from the primary export) update normally.
      try {
        const previouslyCached = await env.SALES_CACHE.get("noon_sales", { type: "json" });
        const cairoNowForNoonStale = new Date(Date.now() + 3 * 60 * 60 * 1000);
        const cairoTodayStrForNoon = cairoNowForNoonStale.toISOString().split("T")[0];
        const prevIsFromToday = previouslyCached && new Date(previouslyCached.fetchedAt + 3 * 60 * 60 * 1000).toISOString().split("T")[0] === cairoTodayStrForNoon;
        if (previouslyCached) {
          // Only trust cached "today" figure if it's actually from today - otherwise zero it
          // rather than leaking yesterday's number forward.
          noonData.todayNoonSales = prevIsFromToday ? (previouslyCached.todayNoonSales || 0) : 0;
          noonData.todayNoonOrders = previouslyCached.todayNoonOrders || 0;
          noonData.recentRealSales = previouslyCached.recentRealSales || 0;
          noonData.recentRealOrders = previouslyCached.recentRealOrders || 0;
          noonData.recentRealDate = previouslyCached.recentRealDate || null;
          noonData.recentRealIsEstimate = previouslyCached.recentRealIsEstimate || false;
          noonData.recentRealSalesSoFar = previouslyCached.recentRealSalesSoFar || 0;
        }
      } catch (e) {
        // If we can't even read the old cache, just proceed with the zeros
      }
    }

    noonData.fetchedAt = Date.now();
    await env.SALES_CACHE.put("noon_sales", JSON.stringify(noonData));
    console.log("Noon cache refreshed:", JSON.stringify(noonData));
  } catch (e) {
    console.log("Noon cache refresh failed:", e.message);
    // Leave the old cached value in place rather than wiping it on a transient failure
  }
}

async function hashPin(pin) {
  const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(String(pin)));
  return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2, "0")).join("");
}
function hrAudit(data, actor, action, detail) {
  data.auditLog.unshift({ id: crypto.randomUUID(), actor, action, detail, timestamp: new Date().toISOString() });
}

  const e = (data.employees || []).find(e => e.id === id);
  return e ? e.name : (id || "Unknown");
}

  return new Response(JSON.stringify(obj), { status, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });
}

          inventoryItem(id: $inventoryItemId) {
            inventoryLevel(locationId: $locationId) {
              quantities(names: ["available"]) {
                quantity
              }
            }
          }
        }`;

       // --- ROUTE: SWITCH UBER ORDER TO NORMAL DELIVERY (fixed 67 EGP, "Cairo and Giza" shipping line) ---
      // The REST orders.json PUT endpoint silently ignores shipping_lines changes on existing

     // --- ROUTE: BOSTA SHIP ORDER (real order -> real delivery, one press from the app) ---

        const lastName = order.customer?.lastName || (addr.name || "").split(" ").slice(1).join(" ") || order.name;

        if (!phone) {
          return new Response(JSON.stringify({ error: "Order has no phone number on file - Bosta requires one" }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });
        }

        const descParts = [];
        let itemsCount = 0;
        order.lineItems.edges.forEach(e => {
          const n = e.node;
          const bundleContents = (n.customAttributes || []).filter(a => a.value && a.value.trim());
          if (bundleContents.length > 0) {
            const counts = {};
            bundleContents.forEach(c => { const name = c.value.trim(); counts[name] = (counts[name] || 0) + 1; });
            descParts.push(`${n.title}:`);
            Object.keys(counts).forEach(name => descParts.push(`${counts[name] * n.quantity}x ${name}`));
            itemsCount += n.quantity * bundleContents.length;
          } else {
            descParts.push(`${n.quantity}x ${n.title}`);
            itemsCount += n.quantity;
          }
        });
        const itemsDescription = descParts.join("\n").slice(0, 250);
        const isCod = order.displayFinancialStatus !== "PAID";
        const codAmount = isCod ? parseFloat(order.totalPriceSet?.shopMoney?.amount || 0) : 0;

          if (!res.ok) {
            return new Response(JSON.stringify({ error: "Bosta rejected the delivery", detail: parsed }), { status: res.status, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });

          const trackingNumber = delivery?.trackingNumber || null;
          const deliveryId = delivery?._id || delivery?.id || null;

                userErrors { field message }

              fulfillment: {
                lineItemsByFulfillmentOrder: [{ fulfillmentOrderId: openFulfillmentOrder.node.id }],

              const fResult = fulfillJson.data?.fulfillmentCreateV2;
              fulfillmentResult = fResult?.userErrors?.length ? { error: fResult.userErrors.map(e => e.message).join(", ") } : { status: fResult?.fulfillment?.status || "unknown" };

          } else {

            await fetch(`https://${SHOPIFY_STORE}/admin/api/2026-07/graphql.json`, {
              method: "POST", headers: { "Content-Type": "application/json", "X-Shopify-Access-Token": SHOPIFY_TOKEN },
              body: JSON.stringify({ query: tagMutation, variables: { id: orderId, tags: ["Sent"] } }),
            });
          } catch (e) { /* tagging is a nice-to-have indicator - don't fail the shipment over it */ }

          return new Response(JSON.stringify({
            status: "created",
            deliveryId,
            trackingNumber,
            fulfillment: fulfillmentResult,
            sentPayload: payload,
          }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });

          return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });

      }

      // --- ROUTE: BOSTA CANCEL ORDER (real cancellation of a delivery created via /bosta-ship-order) ---

            tried.push({ label: attempt.label, error: e.message });
          }

// This month's units ordered per seller SKU, for the Amazon Stock screen's "Sales" sort.
async function getAmazonMonthlyUnitsBySku(env, accessToken) {
  // Real rolling last-30-days units per SKU, sourced from Amazon's own GET_SALES_AND_TRAFFIC_REPORT
  // (asinGranularity SKU aggregates unitsOrdered for the whole requested window per SKU) - this
  // is the same figure Seller Central's own inventory page shows, unlike the old incremental
  // order-item scan which converged slowly and could badly undercount on a busy catalog.
  // Reports take a few minutes, so this refreshes asynchronously: a stale cached result (up to
  // 24h old) is returned immediately while a new report is generated in the background; once

              headers: { "x-amz-access-token": accessToken, "Content-Type": "application/json" },

          return new Response(JSON.stringify({ status: res.status, ok: res.ok, body: json, sentPayload: payload }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });
        } catch (e) {
          return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });

          return new Response(JSON.stringify({ status: res.status, ok: res.ok, body: json, sentPayload: payload }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });

            nextToken = json.pagination?.nextToken || null;

          return new Response(JSON.stringify({ status: lastStatus, ok: lastOk, hasPartneredCarrierOption: hasPCP, body: lastOk ? { transportationOptions: allOptions, pagination: {} } : lastJson }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });

        }

      // --- ROUTE: AMAZON CONFIRM TRANSPORTATION OPTION ---
      if (url.pathname === "/amazon-confirm-transportation-option" || url.pathname === "/amazon-confirm-transportation-option/") {
        try {

// This month's units sold per partner SKU, for the Noon Stock screen's "Sales" sort - same
// sales/views report used for pricing elsewhere, just summed by partner SKU. Cached separately
// from the stock levels themselves so it can be refreshed independently.
// NOTE: this report's Partner_SKU values are free-text and lowercase (e.g. "purple iridescent"),
// while the inventory-ledger report's partner_sku values are compact codes (e.g.
// "PURPLEIRIDESCENT14ML") - so both sides are normalized to compact-alnum-lowercase before matching.
// The sales report's Partner_SKU is often a shorter/looser version of the inventory ledger's
// partner_sku, and the same product can appear under a few slightly different spellings across
// rows (e.g. "cherry tint" and "cherry tint 10ml") - so this sums every units-report key that's
// an exact match OR a substantial (5+ char) substring match. Size (e.g. "7ml"/"14ml") is checked
// separately and must agree between both sides when either one specifies it - otherwise a size
// variant (e.g. "...7ml") would wrongly absorb units meant for the base/other-size product.
function extractSizeMl(s) {
  const m = (s || "").match(/(\d+)ml/);
  return m ? m[1] : null;
}
function lookupUnitsForPartnerSku(inventoryPartnerSku, unitsBySku) {
  const target = compactAlnumWorker(inventoryPartnerSku);
  if (!target) return 0;
  const targetSize = extractSizeMl(target);
  let total = 0;
  for (const key in unitsBySku) {
    const keySize = extractSizeMl(key);
    if (targetSize !== keySize) continue; // require the same explicit size (or both unspecified) - prevents a sized variant (e.g. "...7ml") from absorbing the base/other-size product's sales, or vice versa
    if (key === target) { total += unitsBySku[key]; continue; }
    const shorter = Math.min(target.length, key.length);
    if (shorter >= 5 && (target.includes(key) || key.includes(target))) total += unitsBySku[key];
  }
  return total;
}
async function getNoonMonthlyUnitsBySku(env, forceRefresh) {

  // month-to-date - so this figure is directly comparable to what's shown in Seller Center.

  const windowStart = new Date(cairoNow.getTime() - 30 * 24 * 60 * 60 * 1000);

  const csvText = await noonRunExport(env, "noon_catalog_reports_productviewsandsalesdata", {

    lang: "en",
    from_date: windowStart.toISOString().split("T")[0],
    to_date: cairoNow.toISOString().split("T")[0],
  }, 16, cookie);

  const partnerSkuIdx = header.indexOf("Partner_SKU");
  const shippedUnitsIdx = header.indexOf("Shipped_Units");
  const unitsBySku = {};
  for (let i = 1; i < lines.length; i++) {

  }

              const dLat = (body.lat - shop.lat) * Math.PI / 180;
              const dLng = (body.lng - shop.lng) * Math.PI / 180;
              const a = Math.sin(dLat / 2) ** 2 + Math.cos(shop.lat * Math.PI / 180) * Math.cos(body.lat * Math.PI / 180) * Math.sin(dLng / 2) ** 2;
              const distance = R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
              if (distance <= shop.radiusMeters) { withinAny = true; break; }
              closestDistance = Math.min(closestDistance, distance);
            }
            if (!withinAny) return hrJsonResponse({ error: `You're ${Math.round(closestDistance)}m from the nearest shop - must be within range to check in` }, 400);
          }
          const todayStr = now.split("T")[0];
          const already = hr.attendance.find(a => a.employeeId === body.employeeId && a.date === todayStr && !a.checkOut);
          if (already) return hrJsonResponse({ error: "Already checked in today" }, 400);
          const isFirstToday = !hr.attendance.some(a => a.date === todayStr);
          const record = { id: crypto.randomUUID(), employeeId: body.employeeId, date: todayStr, checkIn: now, checkOut: null, hoursWorked: null, late: body.late || false, lateMinutes: 0, lateDeduction: 0, createdAt: now };
          const emp = hr.employees.find(e => e.id === body.employeeId);
          if (emp && emp.scheduleStart) {
            const [schedH, schedM] = emp.scheduleStart.split(":").map(Number);
            const cairoNowForLate = new Date(Date.now() + 3 * 60 * 60 * 1000);
            const scheduledMinutes = schedH * 60 + schedM;
            const actualMinutes = cairoNowForLate.getUTCHours() * 60 + cairoNowForLate.getUTCMinutes();
            const lateMinutes = Math.max(0, actualMinutes - scheduledMinutes);
            record.lateMinutes = lateMinutes;
            record.lateDeduction = (emp.hourlyRate && !emp.fixedSalary && !emp.commissionBased) ? Math.round((lateMinutes / 60) * emp.hourlyRate * 100) / 100 : 0;
            record.late = lateMinutes > 0;
          }
          hr.attendance.unshift(record);
          hrAudit(hr, body.actor || empName(hr, body.employeeId), "check_in", record.late ? `Checked in ${todayStr}, ${record.lateMinutes}min late (-${record.lateDeduction} EGP)` : `Checked in ${todayStr}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ record, isFirstToday });
        }
        if (url.pathname === "/hr/attendance/checkout" && request.method === "POST") {
          const body = await request.json();
          const shopsOut = hr.settings.shops || [];
          if (shopsOut.length) {
            if (body.lat == null || body.lng == null) return hrJsonResponse({ error: "Location required - enable location access and try again" }, 400);
            const R = 6371000;
            let withinAny = false, closestDistance = Infinity;
            for (const shop of shopsOut) {
              const dLat = (body.lat - shop.lat) * Math.PI / 180;
              const dLng = (body.lng - shop.lng) * Math.PI / 180;
              const a = Math.sin(dLat / 2) ** 2 + Math.cos(shop.lat * Math.PI / 180) * Math.cos(body.lat * Math.PI / 180) * Math.sin(dLng / 2) ** 2;
              const distance = R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
              if (distance <= shop.radiusMeters) { withinAny = true; break; }
              closestDistance = Math.min(closestDistance, distance);
            }
            if (!withinAny) return hrJsonResponse({ error: `You're ${Math.round(closestDistance)}m from the nearest shop - must be within range to check out` }, 400);
          }
          const record = hr.attendance.find(a => a.employeeId === body.employeeId && !a.checkOut);
          if (!record) return hrJsonResponse({ error: "No open check-in found" }, 400);
          record.checkOut = now;
          record.hoursWorked = Math.round(((new Date(record.checkOut) - new Date(record.checkIn)) / 3600000) * 100) / 100;
          const empForCheckout = hr.employees.find(e => e.id === record.employeeId);
          if (empForCheckout && empForCheckout.scheduleEnd) {
            const cairoNowForEarly = new Date(new Date(now).getTime() + 3 * 3600000);
            const [schedEndH, schedEndM] = empForCheckout.scheduleEnd.split(":").map(Number);
            const scheduledEndMinutes = schedEndH * 60 + schedEndM;
            const actualEndMinutes = cairoNowForEarly.getUTCHours() * 60 + cairoNowForEarly.getUTCMinutes();
            const earlyMinutes = Math.max(0, scheduledEndMinutes - actualEndMinutes);
            record.earlyMinutes = earlyMinutes;
            record.earlyDeduction = (empForCheckout.hourlyRate && !empForCheckout.fixedSalary && !empForCheckout.commissionBased) ? Math.round((earlyMinutes / 60) * empForCheckout.hourlyRate * 100) / 100 : 0;
            record.earlyLeave = earlyMinutes > 0;
          }
          hrAudit(hr, body.actor || empName(hr, body.employeeId), "check_out", record.earlyLeave ? `Checked out, ${record.hoursWorked}h (${record.earlyMinutes}min early, -${record.earlyDeduction} EGP)` : `Checked out, ${record.hoursWorked}h`);
          await saveHrData(env, hr);
          return hrJsonResponse({ record });
        }

        // --- Leave requests ---
        if (url.pathname === "/hr/leave" && request.method === "GET") {
          const employeeId = url.searchParams.get("employeeId");
          const list = employeeId ? hr.leaveRequests.filter(l => l.employeeId === employeeId) : hr.leaveRequests;
          return hrJsonResponse({ leaveRequests: list });
        }
        if (url.pathname === "/hr/leave" && request.method === "POST") {
          const body = await request.json();
          if (!body.employeeId || !body.startDate || !body.endDate) return hrJsonResponse({ error: "Missing required fields" }, 400);
          const request_ = { id: crypto.randomUUID(), employeeId: body.employeeId, type: body.type || "annual", startDate: body.startDate, endDate: body.endDate, reason: body.reason || "", status: "pending", createdAt: now, decidedAt: null, decidedBy: null , ownerNotified: false };
          hr.leaveRequests.unshift(request_);
          hrAudit(hr, body.actor || empName(hr, body.employeeId), "leave_requested", `${request_.type} leave ${request_.startDate}-${request_.endDate}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ leaveRequest: request_ });
        }
        if (url.pathname === "/hr/leave/decide" && request.method === "POST") {
          const body = await request.json();
          const lr = hr.leaveRequests.find(l => l.id === body.id);
          if (!lr) return hrJsonResponse({ error: "Leave request not found" }, 404);
          lr.status = body.decision === "approve" ? "approved" : "rejected";
          lr.decidedAt = now; lr.decidedBy = body.actor || "owner"; lr.notified = false;
          if (lr.status === "approved") {
            const emp = hr.employees.find(e => e.id === lr.employeeId);
            if (emp && lr.type in emp.leaveBalance) {
              const days = Math.round((new Date(lr.endDate) - new Date(lr.startDate)) / 86400000) + 1;
              emp.leaveBalance[lr.type] = Math.max(0, (emp.leaveBalance[lr.type] || 0) - days);
            }
          }
          hrAudit(hr, body.actor || "owner", "leave_decided", `${lr.status} for ${empName(hr, lr.employeeId)}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ leaveRequest: lr });
        }
        if (url.pathname === "/hr/leave/ack" && request.method === "POST") {
          const body = await request.json();
          const lr = hr.leaveRequests.find(l => l.id === body.id);
          if (lr) lr.notified = true;
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }
        if (url.pathname === "/hr/leave/owner-ack" && request.method === "POST") {
          const body = await request.json();
          const lr = hr.leaveRequests.find(l => l.id === body.id);
          if (lr) lr.ownerNotified = true;
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }

        // --- Loans / salary advances ---
        if (url.pathname === "/hr/loans" && request.method === "GET") {
          const employeeId = url.searchParams.get("employeeId");
          const list = employeeId ? hr.loans.filter(l => l.employeeId === employeeId) : hr.loans;
          return hrJsonResponse({ loans: list });
        }
        if (url.pathname === "/hr/loans" && request.method === "POST") {
          const body = await request.json();
          if (!body.employeeId || !body.amount) return hrJsonResponse({ error: "Missing required fields" }, 400);
          const isSelfRequest = !!body.selfRequested;
          const loan = { id: crypto.randomUUID(), employeeId: body.employeeId, type: body.type || "loan", amount: parseFloat(body.amount), installments: parseInt(body.installments, 10) || 1, remainingBalance: parseFloat(body.amount), status: isSelfRequest ? "pending" : "active", createdAt: now, decidedAt: isSelfRequest ? null : now, payments: [], notified: !isSelfRequest, ownerNotified: !isSelfRequest };
          hr.loans.unshift(loan);
          hrAudit(hr, body.actor || "owner", isSelfRequest ? "loan_requested" : "loan_created", `${loan.type} of ${loan.amount} for ${empName(hr, loan.employeeId)}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ loan });
        }
        if (url.pathname === "/hr/loans/decide" && request.method === "POST") {
          const body = await request.json();
          const loan = hr.loans.find(l => l.id === body.id);
          if (!loan) return hrJsonResponse({ error: "Loan not found" }, 404);
          loan.status = body.decision === "approve" ? "active" : "rejected";
          loan.decidedAt = now;
          loan.notified = false;
          hrAudit(hr, body.actor || "owner", "loan_decided", `${loan.status} for ${empName(hr, loan.employeeId)}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ loan });
        }
        if (url.pathname === "/hr/loans/ack" && request.method === "POST") {
          const body = await request.json();
          const loan = hr.loans.find(l => l.id === body.id);
          if (loan) loan.notified = true;
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }
        if (url.pathname === "/hr/loans/owner-ack" && request.method === "POST") {
          const body = await request.json();
          const loan = hr.loans.find(l => l.id === body.id);
          if (loan) loan.ownerNotified = true;
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }
        if (url.pathname === "/hr/loans/payment" && request.method === "POST") {
          const body = await request.json();
          const loan = hr.loans.find(l => l.id === body.id);
          if (!loan) return hrJsonResponse({ error: "Loan not found" }, 404);
          const amount = parseFloat(body.amount);
          loan.remainingBalance = Math.max(0, loan.remainingBalance - amount);
          loan.payments.push({ amount, date: now });
          if (loan.remainingBalance === 0) loan.status = "settled";
          hrAudit(hr, body.actor || "owner", "loan_payment", `Paid ${amount} on loan ${loan.id}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ loan });
        }

        if (url.pathname === "/hr/loans" && request.method === "DELETE") {
          const id = url.searchParams.get("id");
          const idx = hr.loans.findIndex(l => l.id === id);
          if (idx === -1) return hrJsonResponse({ error: "Loan not found" }, 404);
          const [removed] = hr.loans.splice(idx, 1);
          hrAudit(hr, url.searchParams.get("actor") || "owner", "loan_deleted", `Deleted ${removed.type} of ${removed.amount} for ${empName(hr, removed.employeeId)}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }

        // --- TEMPORARY: seed test attendance records with arbitrary dates (for demo/testing) ---
        if (url.pathname === "/hr/seed-attendance" && request.method === "POST") {
          const body = await request.json();
          const records = (body.records || []).map(r => ({ id: crypto.randomUUID(), employeeId: body.employeeId, date: r.date, checkIn: r.checkIn, checkOut: r.checkOut, hoursWorked: r.hoursWorked, late: r.late || false, lateMinutes: r.lateMinutes || 0, lateDeduction: r.lateDeduction || 0, createdAt: r.checkIn }));
          hr.attendance.push(...records);
          hrAudit(hr, body.actor || "owner", "attendance_seeded", `Seeded ${records.length} records for ${empName(hr, body.employeeId)}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true, count: records.length });
        }

        // --- Bonuses (owner-initiated, immediate - no approval step) ---
        if (url.pathname === "/hr/bonuses" && request.method === "GET") {
          const employeeId = url.searchParams.get("employeeId");
          const list = employeeId ? hr.bonuses.filter(b => b.employeeId === employeeId) : hr.bonuses;
          return hrJsonResponse({ bonuses: list });
        }
        if (url.pathname === "/hr/bonuses" && request.method === "POST") {
          const body = await request.json();
          if (!body.employeeId || !body.amount) return hrJsonResponse({ error: "Missing required fields" }, 400);
          const bonus = { id: crypto.randomUUID(), employeeId: body.employeeId, amount: parseFloat(body.amount), reason: body.reason || "", createdAt: now, seen: false };
          hr.bonuses.unshift(bonus);
          hrAudit(hr, body.actor || "owner", "bonus_given", `${bonus.amount} EGP bonus for ${empName(hr, bonus.employeeId)}${bonus.reason ? ` (${bonus.reason})` : ""}`);
          await saveHrData(env, hr);
          return hrJsonResponse({ bonus });
        }
        if (url.pathname === "/hr/bonuses" && request.method === "DELETE") {
          const id = url.searchParams.get("id");
          hr.bonuses = hr.bonuses.filter(b => b.id !== id);
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }
        if (url.pathname === "/hr/bonuses/ack" && request.method === "POST") {
          const body = await request.json();
          const bonus = hr.bonuses.find(b => b.id === body.id);
          if (bonus) bonus.seen = true;
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }

        // --- Audit log (read-only view, but owner can clear entries) ---
        if (url.pathname === "/hr/audit-log" && request.method === "GET") {
          return hrJsonResponse({ auditLog: hr.auditLog.slice(0, 300) });
        }
        if (url.pathname === "/hr/audit-log" && request.method === "DELETE") {
          const id = url.searchParams.get("id");
          if (id) {
            hr.auditLog = hr.auditLog.filter(a => a.id !== id);
          } else {
            hr.auditLog = [];
          }
          await saveHrData(env, hr);
          return hrJsonResponse({ ok: true });
        }

        return hrJsonResponse({ error: "Unknown HR route" }, 404);
      }
      // ================== END HR MANAGEMENT ROUTES ==================

      return new Response("Not Found", { status: 404 });
    } catch (e) {
      return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } });
    }
  },
};
